```kotlin
// File: cart-service-shared/src/main/kotlin/com/yourcompany/cart/shared/infrastructure/db/JooqCartsRepository.kt
package com.yourcompany.cart.shared.infrastructure.db

import com.yourcompany.cart.shared.domain.*
import com.yourcompany.cart.shared.infrastructure.db.cs.tables.Cart.CART
import com.yourcompany.cart.shared.infrastructure.db.cs.tables.records.CartRecord
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import kotlinx.serialization.encodeToString
import kotlinx.serialization.json.Json
import org.jooq.DSLContext
import org.jooq.JSON
import org.jooq.JSONB
import org.jooq.impl.DSL
import org.slf4j.LoggerFactory
import java.time.OffsetDateTime
import java.util.UUID

class JooqCartsRepository(
    private val dsl: DSLContext,
    private val json: Json
) : CartRepository {

    private val logger = LoggerFactory.getLogger(javaClass)

    override suspend fun findById(id: UUID): Cart? = withContext(Dispatchers.IO) {
        logger.debug("Finding cart by id: $id")

        dsl.selectFrom(CART)
            .where(CART.ID.eq(id))
            .fetchOne()
            ?.toDomain()
    }

    override suspend fun findByCustomerId(customerId: UUID): List<Cart> = withContext(Dispatchers.IO) {
        logger.debug("Finding carts for customer: $customerId")

        dsl.selectFrom(CART)
            .where(CART.CUSTOMER_ID.eq(customerId))
            .orderBy(CART.CREATED_AT.desc())
            .fetch()
            .map { it.toDomain() }
    }

    override suspend fun findByCustomerIdAndState(
        customerId: UUID,
        state: CartState
    ): Cart? = withContext(Dispatchers.IO) {
        logger.debug("Finding cart for customer: $customerId with state: $state")

        dsl.selectFrom(CART)
            .where(
                CART.CUSTOMER_ID.eq(customerId)
                    .and(CART.STATE.eq(state.name.lowercase()))
            )
            .fetchOne()
            ?.toDomain()
    }

    override suspend fun save(cart: Cart): Cart = withContext(Dispatchers.IO) {
        logger.debug("Saving cart: ${cart.id}")

        val record = CartRecord(
            id = cart.id,
            customerId = cart.customerId,
            orderType = cart.orderType.toJooq(),
            state = cart.state.name.lowercase(),
            items = JSON.valueOf(json.encodeToString(cart.items)),
            metadata = cart.metadata?.let { JSONB.valueOf(json.encodeToString(it)) },
            createdAt = cart.createdAt,
            updatedAt = cart.updatedAt
        )

        dsl.insertInto(CART)
            .set(record)
            .returning()
            .fetchOne()!!
            .toDomain()
    }

    override suspend fun update(cart: Cart): Cart = withContext(Dispatchers.IO) {
        logger.debug("Updating cart: ${cart.id}")

        dsl.update(CART)
            .set(CART.CUSTOMER_ID, cart.customerId)
            .set(CART.ORDER_TYPE, cart.orderType.toJooq())
            .set(CART.STATE, cart.state.name.lowercase())
            .set(CART.ITEMS, JSON.valueOf(json.encodeToString(cart.items)))
            .set(CART.METADATA, cart.metadata?.let { JSONB.valueOf(json.encodeToString(it)) })
            .set(CART.UPDATED_AT, OffsetDateTime.now())
            .where(CART.ID.eq(cart.id))
            .returning()
            .fetchOne()!!
            .toDomain()
    }

    override suspend fun delete(id: UUID): Boolean = withContext(Dispatchers.IO) {
        logger.debug("Deleting cart: $id")

        val deleted = dsl.deleteFrom(CART)
            .where(CART.ID.eq(id))
            .execute()

        deleted > 0
    }

    override suspend fun existsByCustomerIdAndState(
        customerId: UUID,
        state: CartState
    ): Boolean = withContext(Dispatchers.IO) {
        dsl.fetchExists(
            dsl.selectOne()
                .from(CART)
                .where(
                    CART.CUSTOMER_ID.eq(customerId)
                        .and(CART.STATE.eq(state.name.lowercase()))
                )
        )
    }

    // Advanced query methods
    suspend fun findActiveCarts(limit: Int = 100): List<Cart> = withContext(Dispatchers.IO) {
        dsl.selectFrom(CART)
            .where(CART.STATE.eq("active"))
            .orderBy(CART.UPDATED_AT.desc())
            .limit(limit)
            .fetch()
            .map { it.toDomain() }
    }

    suspend fun countByState(state: CartState): Int = withContext(Dispatchers.IO) {
        dsl.selectCount()
            .from(CART)
            .where(CART.STATE.eq(state.name.lowercase()))
            .fetchOne(0, Int::class.java) ?: 0
    }

    suspend fun findStaleCartsForCleanup(daysOld: Int): List<Cart> = withContext(Dispatchers.IO) {
        val cutoffDate = OffsetDateTime.now().minusDays(daysOld.toLong())

        dsl.selectFrom(CART)
            .where(
                CART.STATE.eq("active")
                    .and(CART.UPDATED_AT.lessThan(cutoffDate))
            )
            .fetch()
            .map { it.toDomain() }
    }

    suspend fun findByCustomerIdWithPagination(
        customerId: UUID,
        offset: Int,
        limit: Int
    ): List<Cart> = withContext(Dispatchers.IO) {
        dsl.selectFrom(CART)
            .where(CART.CUSTOMER_ID.eq(customerId))
            .orderBy(CART.CREATED_AT.desc())
            .offset(offset)
            .limit(limit)
            .fetch()
            .map { it.toDomain() }
    }

    // Mapping functions
    private fun CartRecord.toDomain(): Cart {
        return Cart(
            id = id!!,
            customerId = customerId!!,
            orderType = orderType!!.toDomain(),
            state = CartState.valueOf(state!!.uppercase()),
            items = json.decodeFromString(items!!.data()),
            metadata = metadata?.data()?.let { json.decodeFromString(it) },
            createdAt = createdAt!!,
            updatedAt = updatedAt!!
        )
    }

    private fun OrderType.toDomain(): com.yourcompany.cart.shared.domain.OrderType {
        return when (this) {
            com.yourcompany.cart.shared.infrastructure.db.cs.enums.OrderType.planned ->
                com.yourcompany.cart.shared.domain.OrderType.PLANNED
            com.yourcompany.cart.shared.infrastructure.db.cs.enums.OrderType.subscription ->
                com.yourcompany.cart.shared.domain.OrderType.SUBSCRIPTION
            com.yourcompany.cart.shared.infrastructure.db.cs.enums.OrderType.marketplace ->
                com.yourcompany.cart.shared.domain.OrderType.MARKETPLACE
        }
    }

    private fun com.yourcompany.cart.shared.domain.OrderType.toJooq(): com.yourcompany.cart.shared.infrastructure.db.cs.enums.OrderType {
        return when (this) {
            com.yourcompany.cart.shared.domain.OrderType.PLANNED ->
                com.yourcompany.cart.shared.infrastructure.db.cs.enums.OrderType.planned
            com.yourcompany.cart.shared.domain.OrderType.SUBSCRIPTION ->
                com.yourcompany.cart.shared.infrastructure.db.cs.enums.OrderType.subscription
            com.yourcompany.cart.shared.domain.OrderType.MARKETPLACE ->
                com.yourcompany.cart.shared.infrastructure.db.cs.enums.OrderType.marketplace
        }
    }
}
```

```kotlin
class JooqCartsRepository(
    private val dsl: DSLContext,
    private val json: Json
) : CartRepository {

    // Join queries with related tables
    suspend fun findCartWithAuditHistory(cartId: UUID): CartWithHistory? = withContext(Dispatchers.IO) {
        val AUDIT = com.yourcompany.cart.shared.infrastructure.db.cs.tables.CartAuditHistory.CART_AUDIT_HISTORY

        val result = dsl.select(
                CART.asterisk(),
                DSL.jsonArrayAgg(
                    DSL.jsonObject(
                        DSL.key("id").value(AUDIT.ID),
                        DSL.key("action").value(AUDIT.ACTION),
                        DSL.key("changes").value(AUDIT.CHANGES),
                        DSL.key("timestamp").value(AUDIT.CREATED_AT)
                    )
                ).orderBy(AUDIT.CREATED_AT.desc())
            )
            .from(CART)
            .leftJoin(AUDIT).on(AUDIT.CART_ID.eq(CART.ID))
            .where(CART.ID.eq(cartId))
            .groupBy(CART.ID)
            .fetchOne()

        result?.let {
            val cart = it.into(CART).toDomain()
            val history = json.decodeFromString<List<AuditEntry>>(it.get(1, String::class.java))
            CartWithHistory(cart, history)
        }
    }

    // Aggregate queries
    suspend fun getCartStatistics(): CartStatistics = withContext(Dispatchers.IO) {
        val result = dsl.select(
                CART.STATE,
                DSL.count().`as`("count"),
                DSL.avg(DSL.jsonbArrayLength(CART.ITEMS.cast(JSONB::class.java))).`as`("avg_items"),
                DSL.max(CART.CREATED_AT).`as`("latest")
            )
            .from(CART)
            .groupBy(CART.STATE)
            .fetch()

        CartStatistics(
            byState = result.associate {
                it.get(CART.STATE)!! to it.get("count", Int::class.java)!!
            },
            averageItemsPerCart = result.firstOrNull()?.get("avg_items", Double::class.java) ?: 0.0,
            latestCartDate = result.firstOrNull()?.get("latest", OffsetDateTime::class.java)
        )
    }

    // Batch operations
    suspend fun saveBatch(carts: List<Cart>): List<Cart> = withContext(Dispatchers.IO) {
        if (carts.isEmpty()) return@withContext emptyList()

        val records = carts.map { cart ->
            CartRecord(
                id = cart.id,
                customerId = cart.customerId,
                orderType = cart.orderType.toJooq(),
                state = cart.state.name.lowercase(),
                items = JSON.valueOf(json.encodeToString(cart.items)),
                metadata = cart.metadata?.let { JSONB.valueOf(json.encodeToString(it)) },
                createdAt = cart.createdAt,
                updatedAt = cart.updatedAt
            )
        }

        dsl.batchInsert(records)
            .execute()

        carts
    }

    // Conditional updates
    suspend fun updateStateIfCurrent(
        cartId: UUID,
        currentState: CartState,
        newState: CartState
    ): Boolean = withContext(Dispatchers.IO) {
        val updated = dsl.update(CART)
            .set(CART.STATE, newState.name.lowercase())
            .set(CART.UPDATED_AT, OffsetDateTime.now())
            .where(
                CART.ID.eq(cartId)
                    .and(CART.STATE.eq(currentState.name.lowercase()))
            )
            .execute()

        updated > 0
    }

    // Subqueries
    suspend fun findCustomersWithMultipleCarts(): List<UUID> = withContext(Dispatchers.IO) {
        dsl.select(CART.CUSTOMER_ID)
            .from(CART)
            .groupBy(CART.CUSTOMER_ID)
            .having(DSL.count().greaterThan(1))
            .fetch()
            .map { it.get(CART.CUSTOMER_ID)!! }
    }

    // CTE (Common Table Expressions)
    suspend fun findCartsWithRecentActivity(days: Int): List<Cart> = withContext(Dispatchers.IO) {
        val recentCarts = DSL.name("recent_carts")
        val cutoffDate = OffsetDateTime.now().minusDays(days.toLong())

        dsl.with(recentCarts)
            .`as`(
                DSL.select(CART.ID)
                    .from(CART)
                    .where(CART.UPDATED_AT.greaterThan(cutoffDate))
            )
            .select(CART.asterisk())
            .from(CART)
            .join(recentCarts).on(CART.ID.eq(DSL.field(recentCarts.field("id"), UUID::class.java)))
            .fetch()
            .map { it.into(CART).toDomain() }
    }
}
```

### 3. Transaction Management with jOOQ

**Pattern:** Manage database transactions using jOOQ's transaction API with Kotlin coroutines support.

**Example:**
```kotlin
// File: cart-service-shared/src/main/kotlin/com/yourcompany/cart/shared/infrastructure/db/TransactionManager.kt
package com.yourcompany.cart.shared.infrastructure.db

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.jooq.DSLContext
import org.jooq.impl.DSL
import org.slf4j.LoggerFactory
import org.slf4j.MDC

class TransactionManager(private val dsl: DSLContext) {

    private val logger = LoggerFactory.getLogger(javaClass)

    /**
     * Execute a block within a transaction
     * Automatically commits on success, rolls back on exception
     */
    suspend fun <T> transaction(block: suspend (DSLContext) -> T): T = withContext(Dispatchers.IO) {
        val requestId = MDC.get("requestId") ?: "unknown"
        logger.debug("Starting transaction - requestId: $requestId")

        try {
            val result = dsl.transactionResult { configuration ->
                val txDsl = DSL.using(configuration)
                // Note: We need to run the block synchronously within the transaction
                kotlinx.coroutines.runBlocking {
                    block(txDsl)
                }
            }
            logger.debug("Transaction committed successfully - requestId: $requestId")
            result
        } catch (e: Exception) {
            logger.error("Transaction rolled back - requestId: $requestId", e)
            throw e
        }
    }

    /**
     * Execute a block within a transaction without return value
     */
    suspend fun transactionVoid(block: suspend (DSLContext) -> Unit) = withContext(Dispatchers.IO) {
        val requestId = MDC.get("requestId") ?: "unknown"
        logger.debug("Starting void transaction - requestId: $requestId")

        try {
            dsl.transaction { configuration ->
                val txDsl = DSL.using(configuration)
                kotlinx.coroutines.runBlocking {
                    block(txDsl)
                }
            }
            logger.debug("Void transaction committed successfully - requestId: $requestId")
        } catch (e: Exception) {
            logger.error("Void transaction rolled back - requestId: $requestId", e)
            throw e
        }
    }
}
```

**Service Layer with Transactions:**
```kotlin
// File: cart-service-shared/src/main/kotlin/com/yourcompany/cart/shared/service/CartService.kt
package com.yourcompany.cart.shared.service

import com.yourcompany.cart.shared.domain.*
import com.yourcompany.cart.shared.infrastructure.db.JooqCartsRepository
import com.yourcompany.cart.shared.infrastructure.db.TransactionManager
import kotlinx.coroutines.async
import kotlinx.coroutines.coroutineScope
import org.slf4j.LoggerFactory
import java.util.UUID

class CartService(
    private val cartRepository: JooqCartsRepository,
    private val transactionManager: TransactionManager,
    private val eventPublisher: EventPublisher
) {
    private val logger = LoggerFactory.getLogger(javaClass)

    /**
     * Create a new cart and publish event atomically
     */
    suspend fun createCart(customerId: UUID, items: List<CartItem>): Cart {
        return transactionManager.transaction { txDsl ->
            // All operations within this block are in the same transaction
            val cart = Cart.create(
                customerId = customerId,
                items = items
            )

            // Save cart within transaction
            val savedCart = cartRepository.save(cart)

            // Publish event within transaction
            // If event publishing fails, entire transaction rolls back
            eventPublisher.publishCartCreated(savedCart)

            logger.info("Cart created successfully: ${savedCart.id}")
            savedCart
        }
    }

    /**
     * Transfer items from one cart to another atomically
     */
    suspend fun mergeCart(fromCartId: UUID, toCartId: UUID): Cart {
        return transactionManager.transaction { txDsl ->
            // Find both carts within transaction
            val fromCart = cartRepository.findById(fromCartId)
                ?: throw CartNotFoundException(fromCartId)

            val toCart = cartRepository.findById(toCartId)
                ?: throw CartNotFoundException(toCartId)

            // Validate both carts belong to same customer
            require(fromCart.customerId == toCart.customerId) {
                "Cannot merge carts from different customers"
            }

            // Merge items
            val mergedCart = toCart.addItems(fromCart.items)

            // Update target cart
            val updatedCart = cartRepository.update(mergedCart)

            // Delete source cart
            cartRepository.delete(fromCartId)

            logger.info("Carts merged: $fromCartId -> $toCartId")
            updatedCart
        }
    }

    /**
     * Update cart with optimistic locking
     */
    suspend fun updateCart(cartId: UUID, update: (Cart) -> Cart): Cart {
        return transactionManager.transaction { txDsl ->
            // Lock row for update (SELECT ... FOR UPDATE)
            val cart = txDsl.selectFrom(com.yourcompany.cart.shared.infrastructure.db.cs.tables.Cart.CART)
                .where(com.yourcompany.cart.shared.infrastructure.db.cs.tables.Cart.CART.ID.eq(cartId))
                .forUpdate()
                .fetchOne()
                ?.let { cartRepository.mapToDomain(it) }
                ?: throw CartNotFoundException(cartId)

            // Apply update
            val updatedCart = update(cart)

            // Save updated cart
            cartRepository.update(updatedCart)
        }
    }

    /**
     * Batch operation with transaction
     */
    suspend fun createMultipleCarts(requests: List<CreateCartRequest>): List<Cart> {
        return transactionManager.transaction { txDsl ->
            requests.map { request ->
                val cart = Cart.create(
                    customerId = request.customerId,
                    items = request.items
                )
                cartRepository.save(cart)
            }
        }
    }

    /**
     * Complex multi-step operation
     */
    suspend fun checkoutCart(cartId: UUID): Order {
        return transactionManager.transaction { txDsl ->
            // Step 1: Lock and validate cart
            val cart = cartRepository.findById(cartId)
                ?: throw CartNotFoundException(cartId)

            require(cart.state == CartState.ACTIVE) {
                "Cannot checkout cart in state: ${cart.state}"
            }

            require(cart.items.isNotEmpty()) {
                "Cannot checkout empty cart"
            }

            // Step 2: Create order from cart
            val order = Order.fromCart(cart)

            // Step 3: Save order
            val savedOrder = orderRepository.save(order)

            // Step 4: Update cart state
            val completedCart = cart.complete()
            cartRepository.update(completedCart)

            // Step 5: Publish events
            eventPublisher.publishOrderCreated(savedOrder)
            eventPublisher.publishCartCompleted(completedCart)

            logger.info("Cart checked out successfully: $cartId -> order: ${savedOrder.id}")
            savedOrder
        }
    }
}
```

### 6. Connection Pool Configuration

**Pattern:** Configure HikariCP connection pool with appropriate settings for production and test environments.

**Example:**
```kotlin
// File: cart-service-shared/src/main/kotlin/com/yourcompany/cart/shared/infrastructure/db/DataSourceConfig.kt
package com.yourcompany.cart.shared.infrastructure.db

import com.zaxxer.hikari.HikariConfig
import com.zaxxer.hikari.HikariDataSource
import io.ktor.server.config.*
import javax.sql.DataSource

object DataSourceConfig {

    fun create(config: ApplicationConfig): HikariDataSource {
        val hikariConfig = HikariConfig().apply {
            // Connection settings
            jdbcUrl = config.property("database.url").getString()
            username = config.property("database.user").getString()
            password = config.property("database.password").getString()
            driverClassName = "org.postgresql.Driver"

            // Pool sizing
            maximumPoolSize = config.propertyOrNull("database.pool.maxSize")
                ?.getString()?.toInt() ?: 20
            minimumIdle = config.propertyOrNull("database.pool.minIdle")
                ?.getString()?.toInt() ?: 5

            // Connection timeouts
            connectionTimeout = 30000  // 30 seconds
            idleTimeout = 600000       // 10 minutes
            maxLifetime = 1800000      // 30 minutes

            // Connection validation
            connectionTestQuery = "SELECT 1"
            validationTimeout = 5000

            // Performance settings
            isAutoCommit = true
            transactionIsolation = "TRANSACTION_READ_COMMITTED"

            // Connection leak detection
            leakDetectionThreshold = 60000  // 1 minute

            // Pool name for monitoring
            poolName = "CartServicePool"

            // Register metrics
            isRegisterMbeans = true

            // Statement caching
            addDataSourceProperty("cachePrepStmts", "true")
            addDataSourceProperty("prepStmtCacheSize", "250")
            addDataSourceProperty("prepStmtCacheSqlLimit", "2048")
            addDataSourceProperty("useServerPrepStmts", "true")

            // Network timeouts
            addDataSourceProperty("socketTimeout", "30")
            addDataSourceProperty("loginTimeout", "10")

            // PostgreSQL-specific optimizations
            addDataSourceProperty("ApplicationName", "cart-service")
            addDataSourceProperty("reWriteBatchedInserts", "true")

            // Health check
            healthCheckRegistry = null // Configure with Micrometer if needed
        }

        return HikariDataSource(hikariConfig)
    }

    fun createTestDataSource(jdbcUrl: String, username: String, password: String): HikariDataSource {
        val hikariConfig = HikariConfig().apply {
            this.jdbcUrl = jdbcUrl
            this.username = username
            this.password = password
            driverClassName = "org.postgresql.Driver"

            // Smaller pool for tests
            maximumPoolSize = 5
            minimumIdle = 1

            // Faster timeouts for tests
            connectionTimeout = 10000
            idleTimeout = 30000
            maxLifetime = 60000

            connectionTestQuery = "SELECT 1"
            validationTimeout = 3000

            poolName = "CartServiceTestPool"
        }

        return HikariDataSource(hikariConfig)
    }
}
```

```kotlin
// Ensure jOOQ generates before compilation
tasks.named<KotlinCompile>("compileKotlin") {
    dependsOn(tasks.named("generateJooq"))
}
```

**Use Forced Types for Domain Mapping:**

```kotlin
// Map database enums to Kotlin enums
forcedTypes.add(ForcedType().apply {
    userType = "com.yourcompany.cart.shared.domain.OrderType"
    enumConverter = true

```

**Exclude System Tables:**
```kotlin
database {
    includes = ".*"
    excludes = "flyway_schema_history|pg_.*|information_schema.*"

```

### Repository Pattern Guidelines

**Keep Repositories Focused:**
```kotlin
// ✅ Good - Focused on Cart data access
interface CartRepository {
    suspend fun findById(id: UUID): Cart?
    suspend fun save(cart: Cart): Cart
    suspend fun delete(id: UUID): Boolean
}

// ❌ Bad - Mixed responsibilities
interface CartRepository {
    suspend fun findById(id: UUID): Cart?
    suspend fun sendEmailNotification(cart: Cart)  // Wrong layer!
    suspend fun validateCart(cart: Cart): Boolean  // Domain logic!
}
```

**Use Coroutines for Async I/O:**
```kotlin
// ✅ Good - Suspend functions with Dispatchers.IO
override suspend fun findById(id: UUID): Cart? = withContext(Dispatchers.IO) {
    dsl.selectFrom(CART).where(CART.ID.eq(id)).fetchOne()?.toDomain()
}

// ❌ Bad - Blocking without coroutines
override fun findById(id: UUID): Cart? {
    return dsl.selectFrom(CART).where(CART.ID.eq(id)).fetchOne()?.toDomain()
}
```

### Transaction Guidelines

**Use Transactions for Multi-Step Operations:**
```kotlin
// ✅ Good - Atomic operation
suspend fun mergeCart(from: UUID, to: UUID): Cart = transaction { tx ->
    val fromCart = findById(from) ?: throw NotFoundException()
    val toCart = findById(to) ?: throw NotFoundException()

    val merged = toCart.addItems(fromCart.items)
    update(merged)
    delete(from)

    merged
}

// ❌ Bad - Non-atomic, can leave inconsistent state
suspend fun mergeCart(from: UUID, to: UUID): Cart {
    val fromCart = findById(from) ?: throw NotFoundException()
    val toCart = findById(to) ?: throw NotFoundException()

    val merged = toCart.addItems(fromCart.items)
    update(merged)  // If this succeeds...
    delete(from)    // ...but this fails, we have duplicate items!

    return merged
}
```

### Migration Guidelines

**Use Descriptive Migration Names:**

```sql
-- ✅ Good - Clear, descriptive

-- ❌ Bad - Unclear
```

**Always Add Rollback Comments:**

## Anti-Patterns to Avoid

❌ **Don't manually write SQL in strings:**
```kotlin
// ❌ Bad - No type safety, error-prone
val sql = "SELECT * FROM cart WHERE customer_id = ?"
val result = connection.prepareStatement(sql).apply {
    setString(1, customerId.toString())
}.executeQuery()
```

✅ **Use jOOQ DSL:**
```kotlin
// ✅ Good - Type-safe, refactorable
val result = dsl.selectFrom(CART)
    .where(CART.CUSTOMER_ID.eq(customerId))
    .fetch()
```

❌ **Don't expose jOOQ types in domain layer:**
```kotlin
// ❌ Bad - Leaks infrastructure concern
interface CartRepository {
    suspend fun findById(id: UUID): CartRecord  // jOOQ type!
}
```

✅ **Return domain objects:**
```kotlin
// ✅ Good - Domain type
interface CartRepository {
    suspend fun findById(id: UUID): Cart  // Domain type
}
```

❌ **Don't catch and swallow database exceptions:**
```kotlin
// ❌ Bad - Hides errors
suspend fun save(cart: Cart): Cart? {
    return try {
        repository.save(cart)
    } catch (e: Exception) {
        null  // What went wrong? Caller has no idea!
    }
}
```

✅ **Let exceptions propagate or convert to domain exceptions:**
```kotlin
// ✅ Good - Clear error handling
suspend fun save(cart: Cart): Cart {
    return try {
        repository.save(cart)
    } catch (e: DataAccessException) {
        throw RepositoryException("Failed to save cart: ${cart.id}", e)
    }
}
```

❌ **Don't forget to close connections:**
```kotlin
// ❌ Bad - Connection leak
suspend fun getCart(id: UUID): Cart {
    val connection = dataSource.connection
    // If exception occurs, connection never closed!
    val result = connection.prepareStatement("...").executeQuery()
    connection.close()
    return result
}
```

✅ **Use connection pools and let them manage lifecycle:**
```kotlin
// ✅ Good - HikariCP manages connections
suspend fun getCart(id: UUID): Cart = withContext(Dispatchers.IO) {
    dsl.selectFrom(CART).where(CART.ID.eq(id)).fetchOne()?.toDomain()
        ?: throw NotFoundException()
}
```

❌ **Don't run migrations in multiple application instances simultaneously:**
```yaml
# ❌ Bad - Race conditions
deployment:
  replicas: 3  # All 3 try to migrate!

startup:
  command: ["flyway migrate && start-app"]
```

✅ **Run migrations separately or use locking:**
```yaml
# ✅ Good - Separate migration job
kind: Job
metadata:
  name: cart-db-migration
spec:
  template:
    spec:
      containers:
      - name: flyway
        command: ["flyway", "migrate"]
      restartPolicy: Never
---
# Then start application pods
kind: Deployment
metadata:
  name: cart-service
```

❌ **Don't use SELECT * in production code:**
```kotlin
// ❌ Bad - Fetches all columns, breaks when schema changes
dsl.select()
    .from(CART)
    .where(CART.ID.eq(cartId))
    .fetchOne()
```

✅ **Select specific columns:**
```kotlin
// ✅ Good - Explicit columns
dsl.select(
        CART.ID,
        CART.CUSTOMER_ID,
        CART.STATE,
        CART.ITEMS
    )
    .from(CART)
    .where(CART.ID.eq(cartId))
    .fetchOne()
```
